cmake_minimum_required(VERSION 3.20)
project(Revenge VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(revenge_core src/pe.cpp src/pages.cpp src/launch.cpp src/loader.cpp)
target_include_directories(revenge_core PUBLIC include)
add_executable(Revenge src/main.cpp)
target_link_libraries(Revenge PRIVATE revenge_core)
if(WIN32)
  target_sources(Revenge PRIVATE src/gpu_win.cpp src/launch_win.cpp)
  target_compile_definitions(Revenge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(Revenge PRIVATE d3d12 dxgi ole32 shell32 uuid user32)
else()
  target_sources(Revenge PRIVATE src/gpu_stub.cpp)
endif()
if(MSVC)
  target_compile_options(revenge_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(revenge_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_ok)
if(ipo_ok)
  set_property(TARGET Revenge revenge_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(revenge_tests tests/core.cpp)
  target_link_libraries(revenge_tests PRIVATE revenge_core)
  add_test(NAME core COMMAND revenge_tests)
endif()
